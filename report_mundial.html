// login_screen.dart — Handles login with validation & routing
Future<void> _handleLogin() async {
  if (!_formKey.currentState!.validate()) return;

  final authProvider = Provider.of<AuthProvider>(context, listen: false);

  final success = await authProvider.login(
    _emailController.text.trim(),
    _passwordController.text,
  );

  if (success) {
    // Save remembered email if checkbox is checked
    if (_rememberMe) {
      await prefs.setBool(AppConstants.keyRememberMe, true);
      await prefs.setString(AppConstants.keyRememberedEmail, _emailController.text.trim());
    }

    // Route based on email verification status
    if (authProvider.needsEmailVerification) {
      context.go('/email-verification');
    } else {
      context.go(AppRouter.getDashboardRoute(authProvider.userRole!));
    }
  }
}
// auth_provider.dart — Login with lockout protection
Future<bool> login(String email, String password) async {
  // Check if account is locked (5 failed attempts = 15 min lockout)
  if (isLockedOut) {
    _errorMessage = 'Account locked. Try again in $remaining minutes.';
    return false;
  }

  final user = await _authService.signInWithEmailAndPassword(
    email: email, password: password,
  );

  if (user != null) {
    _loginAttempts = 0;  // Reset on success
    _currentUser = user;
    await _saveUserToPrefs();
    await _initializeNotifications();
    return true;
  }
}
// auth_service.dart — Firebase Auth sign-in + Firestore user data
Future<User?> signInWithEmailAndPassword({
  required String email,
  required String password,
}) async {
  final credential = await _auth.signInWithEmailAndPassword(
    email: email, password: password,
  );

  // Update last login timestamp in Firestore
  await _firestore.collection('users').doc(credential.user!.uid)
      .update({'lastLogin': FieldValue.serverTimestamp()});

  // Fetch user profile from Firestore
  final userDoc = await _firestore.collection('users').doc(credential.user!.uid).get();
  return User.fromJson({'id': credential.user!.uid, ...userDoc.data()!});
}

    email: _emailController.text.trim(),
    name: _nameController.text.trim(),
    password: _passwordController.text,
    role: _selectedRole!,   // fan, organizer, security, emergency

    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text('Registration successful!')),
    );
    context.pop(); // Return to login
  }

// auth_service.dart — Creates Firebase Auth + Firestore user document
Future<User?> registerWithEmailAndPassword({...}) async {
  final credential = await _auth.createUserWithEmailAndPassword(
    email: email, password: password,
  );

  // Send email verification
  await credential.user!.sendEmailVerification();

  // Create user profile in Firestore
  await _firestore.collection('users').doc(credential.user!.uid).set({
    'email': email, 'name': name, 'role': role,
    'createdAt': FieldValue.serverTimestamp(),
    'isActive': true, 'emailVerified': false,
  });
}

// auth_provider.dart — Backend toggle
Future<void> toggleLocationSharing() async {
  final newValue = !_currentUser!.locationSharingEnabled;
  if (newValue) {
    final success = await _locationService.startSharing(_currentUser!.id);
    if (!success) { _errorMessage = 'Location permission denied.'; return; }
  } else {
    await _locationService.stopSharing();
  }
  await updateProfile(locationSharingEnabled: newValue);
}

// staff_management_screen.dart — Assign staff to zone
void _showAssignDialog(context, staff) {
  final zones = Provider.of<CrowdProvider>(context).allZones;

  // Zone dropdown from Firestore
  DropdownButton<String>(
    hint: Text('Choose a zone'),
    items: zones.map((zone) {
      final count = staffProvider.getAssignmentsForZone(zone.id).length;
      return DropdownMenuItem(
        value: zone.id,
        child: Text('${zone.name} — $count staff'),
      );
    }).toList(),
  ),

  // Assign button saves to Firestore

    text: 'Assign to Zone',
    onPressed: () async {
      await staffProvider.assignStaffToZone(
        staffId: staff.id, staffName: staff.name,
        zoneId: zone.id, zoneName: zone.name,
        assignedBy: authProvider.currentUser!.id,
      );
    },
  ),
}
